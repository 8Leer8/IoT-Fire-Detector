/*
  =========================================
        IOT FIRE ALERT SYSTEM (ESP32)
     FINAL - APP RESOLVES THE ALARM
     OPTIMIZED - DUAL CORE / FREERTOS
  =========================================

  Core 1 (main loop): sensors, buzzer, LCD — never blocked
  Core 0 (http task): all WiFi/HTTP — blocking is fine here
*/

#include <Wire.h>
#include <LiquidCrystal_I2C.h>
#include <WiFi.h>
#include <WiFiClientSecure.h>
#include <HTTPClient.h>

LiquidCrystal_I2C lcd(0x27, 16, 2);

#define SENSOR_STALL_1  34
#define SENSOR_STALL_2  35
#define BUZZER_PIN      25

const char* WIFI_SSID = "PLDTHOMEFIBRTKdu9";
const char* WIFI_PASS = "PLDTWIFIx5KWJ";

const char* BASE_URL            = "https://iot-fire-detector-production.up.railway.app";
const char* FIRE_ALERT_PATH     = "/api/fire-alert/";
const char* SENSOR_STATUS_PATH  = "/api/sensor-status/";
const char* CHECK_RESOLVED_PATH = "/api/check-resolved/";

// ── Thresholds ────────────────────────────────────────────────────────────────
const int THRESHOLD_ON  = 3300;
const int THRESHOLD_OFF = 3900;

// ── Sensor averaging ──────────────────────────────────────────────────────────
const int     SAMPLES = 15;
long          sampleSum1 = 0, sampleSum2 = 0;
int           sampleCount = 0;
volatile int  avg1 = 4095, avg2 = 4095;
unsigned long lastSampleMs = 0;

// ── Per-stall state ───────────────────────────────────────────────────────────
struct StallState {
  volatile bool fireDetected;
  volatile bool alarmActive;
  volatile bool resolvedByApp;
  volatile unsigned long fireDisappearedTime;
  volatile bool lastFireSent;
};

StallState stall[2] = {
  { false, false, false, 0, false },
  { false, false, false, 0, false }
};

const unsigned long ALARM_CLEAR_DURATION = 10000;

// ── Timers (main loop) ────────────────────────────────────────────────────────
unsigned long lastLcdUpdateMs    = 0;
unsigned long lastWifiCheckMs    = 0;
const unsigned long LCD_INTERVAL_MS  = 50;   // faster LCD = smoother countdown
const unsigned long WIFI_CHECK_MS    = 10000;

// ── HTTP task timing ──────────────────────────────────────────────────────────
const unsigned long HEARTBEAT_INTERVAL_MS  = 10000;
const unsigned long STATUS_INTERVAL_MS     = 30000;
const unsigned long RESOLVE_CHECK_INTERVAL = 3000;
const int           HTTP_TIMEOUT_MS        = 5000;
const int           HTTP_RETRY_COUNT       = 2;
const int           HTTP_FAIL_RECONNECT_THRESHOLD = 3;

// ── FreeRTOS queue (main → http task) ────────────────────────────────────────
struct HttpJob {
  char  url[128];
  char  body[256];
  bool  isGet;
};

QueueHandle_t httpQueue;
SemaphoreHandle_t stallMutex;   // protect shared StallState reads/writes

WiFiClientSecure secureClient;
int consecutiveHttpFails = 0;

// ── Buzzer ────────────────────────────────────────────────────────────────────
int currentBuzzerFreq = -1;

void setBuzzer(int freq) {
  if (currentBuzzerFreq != freq) {
    if (freq > 0) tone(BUZZER_PIN, freq);
    else { noTone(BUZZER_PIN); digitalWrite(BUZZER_PIN, LOW); }
    currentBuzzerFreq = freq;
  }
}

// ── Stall helpers ─────────────────────────────────────────────────────────────
String stallFromFlags(bool s1, bool s2) {
  if (s1 && s2) return "both";
  if (s1)       return "stall_1";
  if (s2)       return "stall_2";
  return "stall_1";
}

void enqueuePost(const String& url, const String& body) {
  HttpJob job;
  url.toCharArray(job.url, sizeof(job.url));
  body.toCharArray(job.body, sizeof(job.body));
  job.isGet = false;
  xQueueSend(httpQueue, &job, 0);  // non-blocking — drop if full
}

void enqueueGet(const String& url) {
  HttpJob job;
  url.toCharArray(job.url, sizeof(job.url));
  job.body[0] = '\0';
  job.isGet = true;
  xQueueSend(httpQueue, &job, 0);
}

// ── HTTP helpers (run on Core 0 only) ────────────────────────────────────────
void handleHttpFail() {
  consecutiveHttpFails++;
  Serial.printf("[NET] Consecutive HTTP fails: %d\n", consecutiveHttpFails);
  if (consecutiveHttpFails >= HTTP_FAIL_RECONNECT_THRESHOLD) {
    Serial.println("[NET] Resetting TLS/WiFi");
    secureClient.stop();
    WiFi.disconnect(false, false);
    vTaskDelay(pdMS_TO_TICKS(200));
    WiFi.reconnect();
    consecutiveHttpFails = 0;
  }
}

String doGet(const char* url) {
  for (int attempt = 1; attempt <= HTTP_RETRY_COUNT; attempt++) {
    HTTPClient http;
    http.begin(secureClient, url);
    http.setTimeout(HTTP_TIMEOUT_MS);
    int code = http.GET();
    if (code == 200) {
      String body = http.getString();
      http.end();
      consecutiveHttpFails = 0;
      return body;
    }
    Serial.printf("[HTTP GET] code=%d attempt=%d url=%s\n", code, attempt, url);
    http.end();
    if (code < 0) secureClient.stop();
    vTaskDelay(pdMS_TO_TICKS(120));
  }
  handleHttpFail();
  return "";
}

bool doPost(const char* url, const char* body) {
  for (int attempt = 1; attempt <= HTTP_RETRY_COUNT; attempt++) {
    Serial.printf("[HTTP POST] -> %s\n", url);
    HTTPClient http;
    http.begin(secureClient, url);
    http.addHeader("Content-Type", "application/json");
    http.setTimeout(HTTP_TIMEOUT_MS);
    int code = http.POST((uint8_t*)body, strlen(body));
    bool ok = (code >= 200 && code < 300);
    Serial.printf("[HTTP] Response: %d (%s) attempt=%d\n", code, ok ? "OK" : "FAIL", attempt);
    http.end();
    if (ok) { consecutiveHttpFails = 0; return true; }
    if (code < 0) secureClient.stop();
    vTaskDelay(pdMS_TO_TICKS(120));
  }
  handleHttpFail();
  return false;
}

// ── HTTP Task (Core 0) ────────────────────────────────────────────────────────
void httpTask(void* pvParams) {
  unsigned long lastHeartbeat    = 0;
  unsigned long lastStatusSend   = 0;
  unsigned long lastResolveCheck = 0;

  for (;;) {
    unsigned long now = millis();

    // WiFi watchdog
    if (now - lastWifiCheckMs >= WIFI_CHECK_MS) {
      if (WiFi.status() != WL_CONNECTED) {
        Serial.println("[WiFi] Reconnecting...");
        WiFi.reconnect();
      }
    }

    // Drain any queued jobs first (fire alerts, sensor status)
    HttpJob job;
    while (xQueueReceive(httpQueue, &job, 0) == pdTRUE) {
      if (WiFi.status() != WL_CONNECTED) break;

      if (job.isGet) {
        String response = doGet(job.url);
        if (response.length() > 0) {
          bool s1res = response.indexOf("\"stall1_resolved\":true") >= 0;
          bool s2res = response.indexOf("\"stall2_resolved\":true") >= 0;

          xSemaphoreTake(stallMutex, portMAX_DELAY);
          if (s1res && stall[0].alarmActive && !stall[0].resolvedByApp) {
            stall[0].resolvedByApp       = true;
            stall[0].fireDisappearedTime = millis();
            Serial.println("[RESOLVE] Stall 1 resolved by app (queued)");
          }
          if (s2res && stall[1].alarmActive && !stall[1].resolvedByApp) {
            stall[1].resolvedByApp       = true;
            stall[1].fireDisappearedTime = millis();
            Serial.println("[RESOLVE] Stall 2 resolved by app (queued)");
          }
          xSemaphoreGive(stallMutex);
        }
      } else {
        doPost(job.url, job.body);
      }
    }

    now = millis();

    // Heartbeat
    if (now - lastHeartbeat >= HEARTBEAT_INTERVAL_MS) {
      lastHeartbeat = now;
      xSemaphoreTake(stallMutex, portMAX_DELAY);
      bool s1 = stall[0].fireDetected;
      bool s2 = stall[1].fireDetected;
      xSemaphoreGive(stallMutex);

      char payload[64];
      snprintf(payload, sizeof(payload),
               "{\"stall1\":%s,\"stall2\":%s}",
               s1 ? "true" : "false",
               s2 ? "true" : "false");
      doPost((String(BASE_URL) + SENSOR_STATUS_PATH).c_str(), payload);
    }

    // Resolve poll (only while unresolved alarm exists)
    xSemaphoreTake(stallMutex, portMAX_DELAY);
    bool s1unresolved = stall[0].alarmActive && !stall[0].resolvedByApp;
    bool s2unresolved = stall[1].alarmActive && !stall[1].resolvedByApp;
    xSemaphoreGive(stallMutex);

    if ((s1unresolved || s2unresolved) &&
        now - lastResolveCheck >= RESOLVE_CHECK_INTERVAL) {
      lastResolveCheck = now;
      String response = doGet((String(BASE_URL) + CHECK_RESOLVED_PATH).c_str());
      if (response.length() > 0) {
        Serial.println("[RESOLVE CHECK] Response: " + response);

        bool s1res = response.indexOf("\"stall1_resolved\":true") >= 0;
        bool s2res = response.indexOf("\"stall2_resolved\":true") >= 0;

        xSemaphoreTake(stallMutex, portMAX_DELAY);
        if (s1res && stall[0].alarmActive && !stall[0].resolvedByApp) {
          stall[0].resolvedByApp       = true;
          stall[0].fireDisappearedTime = millis();
          Serial.println("[RESOLVE] Stall 1 resolved by app");
        }
        if (s2res && stall[1].alarmActive && !stall[1].resolvedByApp) {
          stall[1].resolvedByApp       = true;
          stall[1].fireDisappearedTime = millis();
          Serial.println("[RESOLVE] Stall 2 resolved by app");
        }
        xSemaphoreGive(stallMutex);
      }
    }

    vTaskDelay(pdMS_TO_TICKS(50));  // yield — don't starve Core 0
  }
}

// ══════════════════════════════════════════════════════════════════════════════
void setup() {
  Serial.begin(115200);

  pinMode(SENSOR_STALL_1, INPUT);
  pinMode(SENSOR_STALL_2, INPUT);
  pinMode(BUZZER_PIN, OUTPUT);
  digitalWrite(BUZZER_PIN, LOW);

  secureClient.setInsecure();
  WiFi.setSleep(false);

  lcd.init();
  lcd.backlight();
  lcd.setCursor(0, 0);
  lcd.print(" IOT FIRE ALARM ");

  for (int i = 0; i < 4; i++) {
    lcd.setCursor(0, 1);
    lcd.print(" INITIALIZING...");
    delay(500);
    lcd.setCursor(0, 1);
    lcd.print("                ");
    delay(200);
  }
  lcd.clear();

  WiFi.begin(WIFI_SSID, WIFI_PASS);
  lcd.setCursor(0, 0);
  lcd.print(" CONNECTING...  ");
  Serial.print("[WiFi] Connecting");

  unsigned long start = millis();
  while (WiFi.status() != WL_CONNECTED && millis() - start < 15000) {
    delay(300);
    Serial.print(".");
  }

  if (WiFi.status() == WL_CONNECTED) {
    Serial.println("\n[WiFi] Connected: " + WiFi.localIP().toString());
    lcd.setCursor(0, 1);
    lcd.print(" WIFI CONNECTED ");
  } else {
    Serial.println("\n[WiFi] Failed!");
    lcd.setCursor(0, 1);
    lcd.print(" WIFI FAILED!   ");
  }
  delay(1500);
  lcd.clear();

  // FreeRTOS primitives
  httpQueue  = xQueueCreate(10, sizeof(HttpJob));
  stallMutex = xSemaphoreCreateMutex();

  // HTTP task on Core 0, main loop stays on Core 1
  xTaskCreatePinnedToCore(
    httpTask,    // function
    "httpTask",  // name
    8192,        // stack bytes
    NULL,        // params
    1,           // priority
    NULL,        // handle
    0            // Core 0
  );
}

// ══════════════════════════════════════════════════════════════════════════════
void loop() {
  // Runs on Core 1 — never blocked by HTTP
  unsigned long now = millis();

  // ── Sensor averaging ───────────────────────────────────────────────────────
  if (now - lastSampleMs >= 1) {
    lastSampleMs = now;
    sampleSum1 += analogRead(SENSOR_STALL_1);
    sampleSum2 += analogRead(SENSOR_STALL_2);
    sampleCount++;
    if (sampleCount >= SAMPLES) {
      avg1 = sampleSum1 / SAMPLES;
      avg2 = sampleSum2 / SAMPLES;
      sampleSum1 = sampleSum2 = 0;
      sampleCount = 0;
    }
  }

  // ── Fire detection (hysteresis) ────────────────────────────────────────────
  xSemaphoreTake(stallMutex, portMAX_DELAY);

  int avgs[2] = { avg1, avg2 };
  for (int s = 0; s < 2; s++) {
    if (avgs[s] < THRESHOLD_ON)       stall[s].fireDetected = true;
    else if (avgs[s] > THRESHOLD_OFF) stall[s].fireDetected = false;
  }

  // ── Alarm state machine ────────────────────────────────────────────────────
  for (int s = 0; s < 2; s++) {
    if (stall[s].fireDetected) {
      if (!stall[s].alarmActive) {
        stall[s].alarmActive   = true;
        stall[s].resolvedByApp = false;
        Serial.printf("[STATE] Stall %d alarm activated\n", s + 1);

        // Immediately queue a fire alert
        bool s1f = stall[0].fireDetected;
        bool s2f = stall[1].fireDetected;
        xSemaphoreGive(stallMutex);

        char payload[200];
        snprintf(payload, sizeof(payload),
                 "{\"status\":\"fire\",\"stall\":\"%s\",\"message\":\"Fire detected by ESP32\"}",
                 stallFromFlags(s1f, s2f).c_str());
        enqueuePost(String(BASE_URL) + FIRE_ALERT_PATH, String(payload));

        xSemaphoreTake(stallMutex, portMAX_DELAY);
        stall[s].lastFireSent = true;
      }
    } else {
      if (stall[s].alarmActive && stall[s].resolvedByApp) {
        if (now - stall[s].fireDisappearedTime >= ALARM_CLEAR_DURATION) {
          stall[s].alarmActive   = false;
          stall[s].resolvedByApp = false;
          Serial.printf("[STATE] Stall %d fully cleared\n", s + 1);

          // Queue "normal" alert
          bool s1f = stall[0].fireDetected;
          bool s2f = stall[1].fireDetected;
          xSemaphoreGive(stallMutex);

          char payload[200];
          snprintf(payload, sizeof(payload),
                   "{\"status\":\"normal\",\"stall\":\"%s\",\"message\":\"No fire detected\"}",
                   stallFromFlags(s1f, s2f).c_str());
          enqueuePost(String(BASE_URL) + FIRE_ALERT_PATH, String(payload));

          xSemaphoreTake(stallMutex, portMAX_DELAY);
          stall[s].lastFireSent = false;
          lcd.clear();
        }
      }
    }
  }

  // Snapshot for LCD/buzzer (inside mutex)
  bool s1fire      = stall[0].fireDetected;
  bool s2fire      = stall[1].fireDetected;
  bool s1alarm     = stall[0].alarmActive;
  bool s2alarm     = stall[1].alarmActive;
  bool s1res       = stall[0].resolvedByApp;
  bool s2res       = stall[1].resolvedByApp;
  unsigned long fd0 = stall[0].fireDisappearedTime;
  unsigned long fd1 = stall[1].fireDisappearedTime;

  xSemaphoreGive(stallMutex);

  bool s1unresolved  = s1alarm && !s1res;
  bool s2unresolved  = s2alarm && !s2res;
  bool anyUnresolved = s1unresolved || s2unresolved;
  bool anyResolved   = (s1alarm && s1res) || (s2alarm && s2res);

  // ── Buzzer (zero-blocked, updates every loop tick ~1ms) ───────────────────
  if (anyUnresolved) {
    setBuzzer(1200);
  } else if (anyResolved) {
    // Crisp beep pattern: 100ms on / 400ms off
    setBuzzer((now % 500 < 100) ? 800 : 0);
  } else {
    setBuzzer(0);
  }

  // ── LCD (50ms refresh = smooth countdown, no flicker) ─────────────────────
  if (now - lastLcdUpdateMs >= LCD_INTERVAL_MS) {
    lastLcdUpdateMs = now;

    if (anyUnresolved) {
      lcd.setCursor(0, 0);
      lcd.print(" FIRE DETECTED  ");
      lcd.setCursor(0, 1);
      if (s1unresolved && s2unresolved) lcd.print("S1&S2 SEND ALRT ");
      else if (s1unresolved)            lcd.print("S1 SEND ALERT   ");
      else                              lcd.print("S2 SEND ALERT   ");

    } else if (anyResolved) {
      unsigned long minElapsed = ALARM_CLEAR_DURATION;
      if (s1alarm && s1res) minElapsed = min(minElapsed, now - fd0);
      if (s2alarm && s2res) minElapsed = min(minElapsed, now - fd1);
      int secondsLeft = (int)max(0UL, (ALARM_CLEAR_DURATION - minElapsed) / 1000);

      lcd.setCursor(0, 0);
      lcd.print("  FIRE CLEARED  ");
      char line2[17];
      snprintf(line2, sizeof(line2), " RESET IN: %2ds  ", secondsLeft);
      lcd.setCursor(0, 1);
      lcd.print(line2);

    } else {
      lcd.setCursor(0, 0);
      lcd.print(" AREA MONITORED ");
      lcd.setCursor(0, 1);
      lcd.print(" NO FIRE DETECT ");
    }
  }
}
